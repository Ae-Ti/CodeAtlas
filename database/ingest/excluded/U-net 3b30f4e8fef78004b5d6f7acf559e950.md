```
def readData(datadir, labeldir, H = 512, W = 512):
    img_mtrx  = np.empty(shape = (30, H, W), dtype = np.uint8)
    mask_mtrx = np.empty(shape = (30, H, W))

    # Loop over all image / mask pairs
    for i, (img, lab) in enumerate(
        zip(
            os.listdir(datadir),
            os.listdir(labeldir)
        )
    ):

        # Load image
        image = np.array(
            Image.open(datadir + img)
        )

        # Add channel dimension
        image = np.expand_dims(
            image,
            axis = 0
        )

        # Add to matrix
        img_mtrx[i, :, :] = image

        # Load mask in grayscale (single channel)
        mask = cv2.imread(
            labeldir + lab,
            cv2.IMREAD_GRAYSCALE
        )

        # Binarise
        _, mask = cv2.threshold(
            mask,
            128,
            255,
            cv2.THRESH_BINARY
        )

        # Add to matrix
        mask_mtrx[i, :, :] = mask

    return img_mtrx, mask_mtrx
```

```
class SegmentationDataset(Dataset):
    def __init__(
        self,
        images,
        masks,
        wmap_w0,
        wmap_sigma,
        device,
        transform = None
    ):
        ''' Initialisation function '''

        self.images    = images
        self.masks     = masks
        self.transform = transform
        self.device    = device

        # Parameters for weight map calculation
        self.w0    = wmap_w0
        self.sigma = wmap_sigma

    def __len__(self):
        return len(self.images)

    def __getitem__(self, idx):
        ''' Preprocess and return image, mask, and weight map '''

        image = self.images[idx, :, :]
        mask  = self.masks[idx, :, :]

        if self.transform:

            # Apply transformations
            aug   = self.transform(
                image = image,
                mask  = mask
            )
            image = aug["image"]
            mask  = aug["mask"]

        # Compute weight map
        weights = weight_map(
            mask = mask,
            w0 = self.w0,
            sigma = self.sigma
        )

        # Min-max scale image and mask
        image = self.min_max_scale(
            image,
            min_val = 0,
            max_val = 1
        )
        mask = self.min_max_scale(
            mask,
            min_val = 0,
            max_val = 1
        )

        # Add channel dimensions
        image = np.expand_dims(
            image,
            axis = 0
        )
        weights = np.expand_dims(
            weights,
            axis = 0
        )
        mask = np.expand_dims(
            mask,
            axis = 0
        )

        # Convert to tensors and send to device
        weights = torch.from_numpy(
            weights
        ).double().to(self.device)

        image = torch.from_numpy(
            image
        ).double().to(self.device)

        mask = torch.from_numpy(
            mask
        ).double().to(self.device)

        # Center crop mask and weights
        mask = nn.ZeroPad2d(-94)(mask)
        weights = nn.ZeroPad2d(-94)(weights)

        return image, mask, weights

    @staticmethod
    def min_max_scale(
        image,
        max_val,
        min_val
    ):
        '''Normalization to range of min, max'''

        image_new = (
            (image - np.min(image))
            * (max_val - min_val)
            / (np.max(image) - np.min(image))
            + min_val
        )

        return image_new
```

## 4. Contracting path와 expanding path

### 논문 텍스트

> “The architecture consists of a contracting path to capture context and a symmetric expanding path that enables precise localization.”
> 

### 대응 코드

```
class Unet(nn.Module):
    ''' Unet class
        As suggested in "U-Net: Convolutional Networks for Biomedical Image Segmentation" (https://arxiv.org/pdf/1505.04597.pdf)
    '''

    def __init__(
        self,
        channels,
        no_classes,
        output_size = None
    ):
        '''Initialisation'''

        super().__init__()

        self.output_size = output_size

        # Initialise encoder
        self.encoder = Encoder(channels)

        # Initialise decoder
        dec_channels = list(
            reversed(channels[1:])
        )
        self.decoder = Decoder(dec_channels)

        # Initialise final layer
        self.head = nn.Conv2d(
            in_channels = channels[1],
            out_channels = no_classes,
            kernel_size = 1
        )

    def forward(self, x):
        '''Forward Phase'''

        x = self.encoder(x)
        x = self.decoder(
            x,
            self.encoder.feat_maps
        )
        x = self.head(x)

        # Retain dimensions
        if self.output_size is not None:
            x = F.interpolate(
                x,
                self.output_size
            )

        return x
```

## 5. End-to-end 학습

### 논문 텍스트

> “We show that such a network can be trained end-to-end from very few images.”
> 

### 대응 코드

```
def train(
    model,
    optimizer,
    dataloader,
    criterion,
    effective_batch_size,
    p_bar = None
):
    ''' Training '''

    model.train()
    optimizer.zero_grad()
    running_loss = 0

    for batch_id, (X, y, weights) in enumerate(
        dataloader
    ):

        if p_bar is not None:
            p_bar.set_description_str(
                f'Batch {batch_id + 1}'
            )

        # Forward
        y_hat = model(X)

        # Compute loss
        loss = criterion(
            y,
            y_hat,
            weights
        ) / effective_batch_size

        running_loss += loss.item()

        loss.backward()

        # Backprop
        if (
            (batch_id + 1)
            % effective_batch_size
            == 0
        ) or (
            (batch_id + 1)
            == len(dataloader)
        ):
            optimizer.step()
            optimizer.zero_grad()

        # Update progress bar
        if p_bar is not None:
            p_bar.set_postfix(
                loss = loss.item()
            )
            p_bar.update(1)

    # Compute average loss
    running_loss = (
        running_loss
        / len(dataloader)
        * effective_batch_size
    )

    return running_loss
```

```
from __future__ import annotations

import numpy as np
import torch

def predict_probability_map(
    model: torch.nn.Module,
    image: torch.Tensor
) -> np.ndarray:
    model.eval()

    with torch.no_grad():
        logits = model(image)
        probability = torch.sigmoid(logits)

    return (
        probability
        .squeeze()
        .detach()
        .cpu()
        .numpy()
    )

def pixel_error_at_threshold(
    probability_map: np.ndarray,
    ground_truth: np.ndarray,
    threshold: float
) -> float:
    prediction = (
        probability_map >= threshold
    ).astype(np.uint8)

    target = (
        ground_truth > 0
    ).astype(np.uint8)

    return float(
        np.mean(prediction != target)
    )

def evaluate_ten_thresholds(
    probability_map: np.ndarray,
    ground_truth: np.ndarray
) -> dict[float, float]:
    thresholds = np.linspace(
        0.0,
        1.0,
        num=10
    )

    results: dict[float, float] = {}

    for threshold in thresholds:
        results[float(threshold)] = (
            pixel_error_at_threshold(
                probability_map,
                ground_truth,
                float(threshold)
            )
        )

    return results
```

```
from __future__ import annotations

import statistics
import time

import torch

def benchmark_inference(
    model: torch.nn.Module,
    image: torch.Tensor,
    warmup_runs: int = 10,
    measured_runs: int = 100
) -> dict[str, float]:
    if image.ndim != 4:
        raise ValueError(
            "입력은 [B, C, H, W] 형태여야 합니다."
        )

    device = next(
        model.parameters()
    ).device

    image = image.to(device)
    model.eval()

    with torch.no_grad():
        for _ in range(warmup_runs):
            model(image)

    if device.type == "cuda":
        torch.cuda.synchronize()

    elapsed_times: list[float] = []

    with torch.no_grad():
        for _ in range(measured_runs):
            start = time.perf_counter()

            model(image)

            if device.type == "cuda":
                torch.cuda.synchronize()

            elapsed_times.append(
                time.perf_counter() - start
            )

    mean_seconds = statistics.mean(
        elapsed_times
    )

    return {
        "mean_seconds": mean_seconds,
        "mean_milliseconds": (
            mean_seconds * 1000.0
        ),
        "fps": (
            1.0 / mean_seconds
            if mean_seconds > 0
            else float("inf")
        )
    }
```

```
class EarlyStopping(object):
    '''Early Stopping'''

    def __init__(self, patience, fname):
        self.patience  = patience
        self.best_loss = np.Inf
        self.counter   = 0
        self.filename  = fname

    def __call__(
        self,
        epoch,
        loss,
        optimizer,
        model
    ):

        if loss < self.best_loss:
            self.counter = 0
            self.best_loss = loss

            torch.save(
                {
                    'epoch': epoch,
                    'model_state_dict': (
                        model.state_dict()
                    ),
                    'optimizer_state_dict': (
                        optimizer.state_dict()
                    ),
                    'loss': loss,
                },
                self.filename
            )

        else:
            self.counter += 1

        return (
            self.counter
            == self.patience
        )
```

## 11. Ciresan의 sliding-window 방식

### 논문 텍스트

> “Ciresan et al. trained a network in a sliding-window setup to predict the class label of each pixel by providing a local region (patch) around that pixel as input.”
> 

### 대응 코드

이 방식은 비교 대상이므로 저장소에는 없다.

### (예상코드)

```
from __future__ import annotations

import torch
import torch.nn.functional as F

def sliding_window_pixel_prediction(
    classifier: torch.nn.Module,
    image: torch.Tensor,
    patch_size: int
) -> torch.Tensor:
    if patch_size % 2 == 0:
        raise ValueError(
            "patch_size는 홀수여야 합니다."
        )

    if image.ndim != 4:
        raise ValueError(
            "입력은 [B, C, H, W]여야 합니다."
        )

    padding = patch_size // 2

    padded = F.pad(
        image,
        (
            padding,
            padding,
            padding,
            padding
        ),
        mode="reflect"
    )

    batch_size, _, height, width = (
        image.shape
    )

    output = torch.empty(
        (
            batch_size,
            height,
            width
        ),
        device=image.device
    )

    for y in range(height):
        for x in range(width):
            patch = padded[
                :,
                :,
                y:y + patch_size,
                x:x + patch_size
            ]

            logits = classifier(patch)

            output[:, y, x] = (
                logits.argmax(dim=1)
            )

    return output
```

### 대응 코드

```
import torch
from torch import nn
import torch.nn.functional as F
import numpy as np

class Block(nn.Module):
    ''' One block of Unet.
        Contains 2 repeated 3 x 3 unpadded convolutions, each followed by a ReLU.
    '''

    def __init__(
        self,
        in_channel,
        out_channel,
        kernel_size
    ):
        ''' Initialisation '''

        super().__init__()

        self.conv_1 = nn.Conv2d(
            in_channel,
            out_channel,
            kernel_size
        )

        self.conv_2 = nn.Conv2d(
            out_channel,
            out_channel,
            kernel_size
        )

        self.relu = nn.ReLU()

        # Initialise weights on convolutional layers
        nn.init.normal_(
            self.conv_1.weight,
            mean = 0.0,
            std = self.init_std(
                in_channel,
                kernel_size
            )
        )

        nn.init.normal_(
            self.conv_1.weight,
            mean = 0.0,
            std = self.init_std(
                out_channel,
                kernel_size
            )
        )

    @staticmethod
    def init_std(
        channels,
        kernel_size
    ):
        ''' Computes std for weight initialisation on the convolutional layers'''

        return (
            2.0
            / np.sqrt(
                channels
                * kernel_size ** 2
            )
        )

    def forward(self, x):
        ''' Forward Phase '''

        x = self.conv_1(x)
        x = self.relu(x)

        x = self.conv_2(x)
        x = self.relu(x)

        return x

class Encoder(nn.Module):
    ''' Contractive Part of Unet '''

    def __init__(self, channels):
        '''Initialisation'''

        super().__init__()

        modules = []

        for in_channel, out_channel in zip(
            channels[:-1],
            channels[1:]
        ):
            block = Block(
                in_channel = in_channel,
                out_channel = out_channel,
                kernel_size = 3
            )

            modules.append(block)

        self.blocks = nn.ModuleList(
            modules = modules
        )

        self.max_pol = nn.MaxPool2d(
            kernel_size = 2,
            stride = None
        )

        self.feat_maps = []

    def forward(self, x):
        '''Forward phase'''

        for layer_no, layer in enumerate(
            self.blocks
        ):

            x = layer(x)

            if not self.is_final_layer(
                layer_no
            ):
                self.feat_maps.append(x)
                x = self.max_pol(x)

        return x

    def is_final_layer(
        self,
        layer_no
    ):
        return (
            layer_no
            == len(self.blocks) - 1
        )

class Decoder(nn.Module):
    ''' Expansive Part of Unet '''

    def __init__(self, channels):
        '''Initialisation'''

        super().__init__()

        up_convs = []
        blocks = []

        for in_channel, out_channel in zip(
            channels[:-1],
            channels[1:]
        ):
            upconv = nn.ConvTranspose2d(
                in_channel,
                out_channel,
                kernel_size = 2,
                stride = 2
            )

            up_convs.append(upconv)

            block = Block(
                in_channel,
                out_channel,
                kernel_size = 3
            )

            blocks.append(block)

        self.upconvs = nn.ModuleList(
            up_convs
        )

        self.blocks = nn.ModuleList(
            blocks
        )

    def forward(
        self,
        x,
        encoded_feat_maps
    ):
        for upconv, block in zip(
            self.upconvs,
            self.blocks
        ):
            x = upconv(x)

            fts = encoded_feat_maps.pop()

            fts = self.crop(
                fts,
                x.shape[2],
                x.shape[3]
            )

            x = torch.cat(
                [x, fts],
                dim = 1
            )

            x = block(x)

        return x

    @staticmethod
    def crop(
        tnsr,
        new_H,
        new_W
    ):
        ''' Center crop an input tensor to shape [hew_H, hew_W] '''

        _, _, H, W = tnsr.size()

        x1 = int(
            round(
                (H - new_H)
                / 2.0
            )
        )

        y1 = int(
            round(
                (W - new_W)
                / 2.0
            )
        )

        x2 = x1 + new_H
        y2 = y1 + new_W

        return tnsr[
            :,
            :,
            x1:x2,
            y1:y2
        ]

class Unet(nn.Module):
    ''' Unet class
        As suggested in "U-Net: Convolutional Networks for Biomedical Image Segmentation" (https://arxiv.org/pdf/1505.04597.pdf)
    '''

    def __init__(
        self,
        channels,
        no_classes,
        output_size = None
    ):
        '''Initialisation'''

        super().__init__()

        self.output_size = output_size

        self.encoder = Encoder(
            channels
        )

        dec_channels = list(
            reversed(channels[1:])
        )

        self.decoder = Decoder(
            dec_channels
        )

        self.head = nn.Conv2d(
            in_channels = channels[1],
            out_channels = no_classes,
            kernel_size = 1
        )

    def forward(self, x):
        '''Forward Phase'''

        x = self.encoder(x)

        x = self.decoder(
            x,
            self.encoder.feat_maps
        )

        x = self.head(x)

        if self.output_size is not None:
            x = F.interpolate(
                x,
                self.output_size
            )

        return x
```

## 19. Figure 2의 overlap-tile strategy

### 논문 텍스트

> “Overlap-tile strategy for seamless segmentation of arbitrary large images.”
> 

> “Prediction of the segmentation in the yellow area requires image data within the blue area as input.”
> 

### 대응 코드

저장소에는 overlap-tile 추론이 없다.

### (예상코드)

```
from __future__ import annotations

import math

import numpy as np
import torch

def overlap_tile_predict(
    model: torch.nn.Module,
    image: np.ndarray,
    input_tile_size: int = 572,
    output_tile_size: int = 388,
    device: str = "cuda"
) -> np.ndarray:
    if image.ndim != 2:
        raise ValueError(
            "이 함수는 단일 채널 2차원 이미지를 받습니다."
        )

    context_border = (
        input_tile_size
        - output_tile_size
    ) // 2

    height, width = image.shape

    tile_rows = math.ceil(
        height / output_tile_size
    )

    tile_columns = math.ceil(
        width / output_tile_size
    )

    padded_output_height = (
        tile_rows
        * output_tile_size
    )

    padded_output_width = (
        tile_columns
        * output_tile_size
    )

    extra_bottom = (
        padded_output_height
        - height
    )

    extra_right = (
        padded_output_width
        - width
    )

    padded_image = np.pad(
        image,
        (
            (
                context_border,
                context_border
                + extra_bottom
            ),
            (
                context_border,
                context_border
                + extra_right
            )
        ),
        mode="reflect"
    )

    probability_map = np.zeros(
        (
            padded_output_height,
            padded_output_width
        ),
        dtype=np.float32
    )

    model = model.to(device)
    model.eval()

    with torch.no_grad():
        for tile_row in range(tile_rows):
            for tile_column in range(
                tile_columns
            ):
                output_y = (
                    tile_row
                    * output_tile_size
                )

                output_x = (
                    tile_column
                    * output_tile_size
                )

                tile = padded_image[
                    output_y:
                    output_y + input_tile_size,
                    output_x:
                    output_x + input_tile_size
                ]

                if tile.shape != (
                    input_tile_size,
                    input_tile_size
                ):
                    raise RuntimeError(
                        "입력 타일 크기가 잘못되었습니다."
                    )

                tile_tensor = (
                    torch.from_numpy(
                        tile.astype(
                            np.float32
                        )
                    )
                    .unsqueeze(0)
                    .unsqueeze(0)
                    .to(device)
                )

                logits = model(
                    tile_tensor
                )

                probability = (
                    torch.sigmoid(logits)
                    .squeeze()
                    .cpu()
                    .numpy()
                )

                if probability.shape != (
                    output_tile_size,
                    output_tile_size
                ):
                    raise RuntimeError(
                        "모델 출력 크기가 "
                        f"{probability.shape}이며 "
                        f"예상 크기 "
                        f"{output_tile_size}×"
                        f"{output_tile_size}와 다릅니다."
                    )

                probability_map[
                    output_y:
                    output_y + output_tile_size,
                    output_x:
                    output_x + output_tile_size
                ] = probability

    return probability_map[
        :height,
        :width
    ]
```

## 28. Touching object와 weighted loss

### 논문 텍스트

> “We propose the use of a weighted loss, where the separating background labels between touching cells obtain a large weight in the loss function.”
> 

### 대응 코드

Weight map:

```
def weight_map(
    mask,
    w0,
    sigma,
    background_class = 0
):
    if mask.dtype != 'uint8':
        mask = mask.astype(
            'uint8'
        )

    wc = _class_weights(mask)

    _, regions = cv2.connectedComponents(
        mask
    )

    region_ids = sorted(
        np.unique(regions)
    )

    region_ids = [
        region_id
        for region_id in region_ids
        if region_id != background_class
    ]

    if len(region_ids) > 1:
        distances = np.zeros(
            (
                mask.shape[0],
                mask.shape[1],
                len(region_ids)
            )
        )

        for i, region_id in enumerate(
            region_ids
        ):
            m = (
                regions != region_id
            ).astype(np.uint8)

            distances[:, :, i] = (
                cv2.distanceTransform(
                    m,
                    distanceType = cv2.DIST_L2,
                    maskSize = 0
                )
            )

        distances = np.sort(
            distances,
            axis = 2
        )

        d1 = distances[:, :, 0]
        d2 = distances[:, :, 1]

        w = (
            w0
            * np.exp(
                -1
                / (2 * sigma ** 2)
                * (d1 + d2) ** 2
            )
            * (
                regions
                == background_class
            )
        )

    else:
        w = np.zeros_like(mask)

    wc_x = np.zeros_like(mask)

    for pixel_class, weight in wc.items():
        wc_x[
            mask == pixel_class
        ] = weight

    w = w + wc_x

    return w
```

Loss:

```
class WeightedBCEWithLogitsLoss(nn.Module):
    ''' Pixel-wise weighted BCEWithLogitsLoss'''

    def __init__(
        self,
        batch_size
    ):
        super().__init__()

        self.batch_size = batch_size

        self.unw_loss = (
            nn.BCEWithLogitsLoss(
                reduction = 'none'
            )
        )

    def __call__(
        self,
        true,
        predicted,
        weights
    ):
        loss = self.unw_loss(
            predicted,
            true
        ) * weights

        loss = loss.sum(
            dim = 1
        )

        loss = loss.view(
            self.batch_size,
            -1
        ) / weights.view(
            self.batch_size,
            -1
        )

        loss = loss.mean()

        return loss
```

```
import torch
from torch import nn

class CorrectWeightedBCEWithLogitsLoss(
    nn.Module
):
    def __init__(self):
        super().__init__()

        self.base_loss = (
            nn.BCEWithLogitsLoss(
                reduction="none"
            )
        )

    def forward(
        self,
        predicted: torch.Tensor,
        target: torch.Tensor,
        weight_map: torch.Tensor
    ) -> torch.Tensor:
        if predicted.shape != target.shape:
            raise ValueError(
                "prediction과 target의 shape이 다릅니다."
            )

        if weight_map.shape != target.shape:
            raise ValueError(
                "weight map과 target의 shape이 다릅니다."
            )

        pixel_loss = self.base_loss(
            predicted,
            target
        )

        weighted_pixel_loss = (
            pixel_loss
            * weight_map
        )

        denominator = (
            weight_map.sum()
            .clamp_min(1e-8)
        )

        return (
            weighted_pixel_loss.sum()
            / denominator
        )
```

```
def count_convolutional_layers(

    conv2d_count = 0

    for module in model.modules():
```

## 31. Even-sized pooling input 조건

### 논문 텍스트

> “It is important to select the input tile size such that all 2x2 max-pooling operations are applied to a layer with an even x- and y-size.”
> 

### 대응 코드

저장소에는 입력 크기 검증이 없다.

### (예상코드)

```
from __future__ import annotations

def validate_unet_input_size(
    height: int,
    width: int,
    number_of_pooling_layers: int = 4,
    convolutions_per_block: int = 2,
    kernel_size: int = 3
) -> None:
    current_height = height
    current_width = width

    shrink_per_convolution = (
        kernel_size - 1
    )

    for level in range(
        number_of_pooling_layers
    ):
        current_height -= (
            convolutions_per_block
            * shrink_per_convolution
        )

        current_width -= (
            convolutions_per_block
            * shrink_per_convolution
        )

        if (
            current_height <= 0
            or current_width <= 0
        ):
            raise ValueError(
                f"{level}단계 convolution 이후 "
                "feature map 크기가 0 이하입니다."
            )

        if (
            current_height % 2 != 0
            or current_width % 2 != 0
        ):
            raise ValueError(
                f"{level}단계 pooling 직전 크기 "
                f"{current_height}×{current_width}가 "
                "짝수가 아닙니다."
            )

        current_height //= 2
        current_width //= 2
```

```
import torch
import torch.nn.functional as F

def center_crop_to_prediction(
    tensor: torch.Tensor,
    prediction: torch.Tensor
) -> torch.Tensor:
    target_height = prediction.shape[-2]
    target_width = prediction.shape[-1]

    source_height = tensor.shape[-2]
    source_width = tensor.shape[-1]

    if (
        target_height > source_height
        or target_width > source_width

        raise ValueError(
            "prediction이 target보다 큽니다."
        )

        source_height
        - target_height
    ) // 2

    left = (
        source_width
        - target_width
    ) // 2

    return tensor[
        ...,
        top:top + target_height,
        left:left + target_width
    ]
```

```
from __future__ import annotations

import torch
import torch.nn.functional as F
from torch import nn

class WeightedPixelCrossEntropyLoss(
    nn.Module
):
    def __init__(self):
        super().__init__()

    def forward(
        self,
        logits: torch.Tensor,
        target: torch.Tensor,
        weight_map: torch.Tensor
    ) -> torch.Tensor:
        if logits.ndim != 4:
            raise ValueError(
                "logits는 [B, K, H, W]여야 합니다."
            )

        if target.ndim != 3:
            raise ValueError(
                "target은 [B, H, W]여야 합니다."
            )

        if weight_map.ndim == 4:
            weight_map = weight_map.squeeze(1)

        if weight_map.shape != target.shape:
            raise ValueError(
                "weight_map과 target의 shape이 다릅니다."
            )

        pixel_loss = F.cross_entropy(
            logits,
            target.long(),
            reduction="none"
        )

        weighted_loss = (
            pixel_loss
            * weight_map
        )

        return (
            weighted_loss.sum()
            / weight_map.sum().clamp_min(1e-8)
        )
```

## 39. 클래스 빈도 보정

```
def _class_weights(mask):
    ''' Create a dictionary containing the classes in a mask,
        and their corresponding weights to balance their occurence
    '''

    wc = defaultdict()

    unique, counts = np.unique(
        mask,
        return_counts = True
    )

    counts = (
        counts
        / np.product(mask.shape)
    )

    max_count = max(counts)

    for val, count in zip(
        unique,
        counts
    ):
        wc[val] = (
            max_count
            / count
        )

    return wc
```

```
from __future__ import annotations

import cv2
import numpy as np

def create_separation_border(
    instance_mask: np.ndarray,
    kernel_size: int = 3
) -> np.ndarray:
    if instance_mask.ndim != 2:
        raise ValueError(
            "instance_mask는 2차원이어야 합니다."
        )

    instance_ids = [
        int(value)
        for value in np.unique(
            instance_mask
        )
        if value != 0
    ]

    kernel = np.ones(
        (
            kernel_size,
            kernel_size
        ),
        dtype=np.uint8
    )

    expanded_instances: list[
        np.ndarray
    ] = []

    for instance_id in instance_ids:
        binary_instance = (
            instance_mask
            == instance_id
        ).astype(np.uint8)

        expanded = cv2.dilate(
            binary_instance,
            kernel,
            iterations=1
        )

        expanded_instances.append(
            expanded
        )

    if len(expanded_instances) < 2:
        return np.zeros_like(
            instance_mask,
            dtype=np.uint8
        )

    overlap_count = np.sum(
        np.stack(
            expanded_instances,
            axis=0
        ),
        axis=0
    )

    separation_border = (
        overlap_count >= 2
    ).astype(np.uint8)

    return separation_border
```

```
import math

from torch import nn

def initialize_unet_convolution(
    convolution: nn.Conv2d
) -> None:
    incoming_nodes = (
        convolution.in_channels
        * convolution.kernel_size[0]
        * convolution.kernel_size[1]

        2.0 / incoming_nodes

        mean=0.0,
```

## 52. Contracting path 끝의 dropout

### 논문 텍스트

> “Drop-out layers at the end of the contracting path perform further implicit data augmentation.”
> 

### 대응 코드

저장소에는 dropout이 없다.

### (예상코드)

```
from __future__ import annotations

import torch
from torch import nn

class EncoderWithDropout(nn.Module):
    def __init__(
        self,
        channels: list[int],
        dropout_probability: float = 0.5
    ):
        super().__init__()

        modules: list[nn.Module] = []

        for in_channel, out_channel in zip(
            channels[:-1],
            channels[1:]
        ):
            modules.append(
                Block(
                    in_channel=in_channel,
                    out_channel=out_channel,
                    kernel_size=3
                )
            )

        self.blocks = nn.ModuleList(
            modules
        )

        self.max_pool = nn.MaxPool2d(
            kernel_size=2,
            stride=2
        )

        self.dropout = nn.Dropout2d(
            p=dropout_probability
        )

        self.feature_maps: list[
            torch.Tensor
        ] = []

    def forward(
        self,
        x: torch.Tensor
    ) -> torch.Tensor:
        self.feature_maps = []

        for layer_index, block in enumerate(
            self.blocks
        ):
            x = block(x)

            final_layer = (
                layer_index
                == len(self.blocks) - 1
            )

            if final_layer:
                x = self.dropout(x)

            else:
                self.feature_maps.append(x)
                x = self.max_pool(x)

        return x
```

### (예상코드)

Pixel error는 앞에서 제공했다. Rand error의 간단한 binary pairwise 구현은 다음과 같다.

```
from __future__ import annotations

import numpy as np

def rand_error(
    prediction: np.ndarray,
    target: np.ndarray
) -> float:
    prediction_flat = (
        prediction.reshape(-1)
    )

    target_flat = target.reshape(-1)

    same_prediction = (
        prediction_flat[:, None]
        == prediction_flat[None, :]
    )

    same_target = (
        target_flat[:, None]
        == target_flat[None, :]
    )

    agreement = np.mean(
        same_prediction
        == same_target
    )

    return float(
        1.0 - agreement
    )
```

이 구현은 큰 이미지에서 `O(N²)` 메모리를 사용하므로 실전 challenge 평가에는 적합하지 않다. Warping error는 challenge의 topology-aware 공식 구현을 사용해야 하며, PDF 본문만으로 정확한 구현을 재구성할 수 없다.

```
from __future__ import annotations

import torch
import torch.nn.functional as F

def rotate_tensor(
    image: torch.Tensor,
    angle_degrees: float
) -> torch.Tensor:
    angle_radians = torch.tensor(
        angle_degrees
        * 3.141592653589793
        / 180.0,
        dtype=image.dtype,
        device=image.device
    )

    cosine = torch.cos(
        angle_radians
    )

    sine = torch.sin(
        angle_radians
    )

    theta = torch.zeros(
        (
            image.shape[0],
            2,
            3
        ),
        dtype=image.dtype,
        device=image.device
    )

    theta[:, 0, 0] = cosine
    theta[:, 0, 1] = -sine
    theta[:, 1, 0] = sine
    theta[:, 1, 1] = cosine

    grid = F.affine_grid(
        theta,
        image.size(),
        align_corners=False
    )

    return F.grid_sample(
        image,
        grid,
        mode="bilinear",
        padding_mode="reflection",
        align_corners=False
    )

def rotation_ensemble_predict(
    model: torch.nn.Module,
    image: torch.Tensor,
    angles_degrees: list[float]
) -> torch.Tensor:
    if len(angles_degrees) != 7:
        raise ValueError(
            "논문 조건에 맞추려면 "
            "7개의 회전 각도를 제공해야 합니다."
        )

    predictions: list[
        torch.Tensor
    ] = []

    model.eval()

    with torch.no_grad():
        for angle in angles_degrees:
            rotated_image = rotate_tensor(
                image,
                angle
            )

            rotated_prediction = (
                torch.sigmoid(
                    model(rotated_image)
                )
            )

            restored_prediction = (
                rotate_tensor(
                    rotated_prediction,
                    -angle
                )
            )

            predictions.append(
                restored_prediction
            )

    return torch.stack(
        predictions,
        dim=0
    ).mean(dim=0)
```

### (예상코드)

```
from __future__ import annotations

import torch

def binary_intersection_over_union(
    logits: torch.Tensor,
    target: torch.Tensor,
    threshold: float = 0.5,
    epsilon: float = 1e-7
) -> torch.Tensor:
    probability = torch.sigmoid(
        logits
    )

    prediction = (
        probability >= threshold
    )

    target_binary = (
        target >= 0.5
    )

    reduce_dimensions = tuple(
        range(
            1,
            prediction.ndim
        )
    )

    intersection = (
        prediction
        & target_binary
    ).sum(
        dim=reduce_dimensions
    ).float()

    union = (
        prediction
        | target_binary
    ).sum(
        dim=reduce_dimensions
    ).float()

    iou = (
        intersection + epsilon
    ) / (
        union + epsilon
    )

    return iou.mean()
```

## 69. Learning-rate range test

### 논문 대응 텍스트

논문 본문에는 구체적인 learning rate가 없다.

### 저장소 추가 코드

```
from torch import optim
from tqdm.notebook import tqdm

class LRTest(object):

    def __init__(
        self,
        min_lr,
        max_lr,
        no_iter,
        batch_size
    ):
        ''' Initialisation function '''

        self.batch_size = batch_size
        self.no_iter = no_iter

        self.lr_multiplier = (
            max_lr / min_lr
        ) ** (
            1 / no_iter
        )

        self.dataiter = None

    def __call__(
        self,
        dataloader,
        criterion,
        optimizer,
        model
    ):
        ''' LR Range test '''

        model.train()

        scheduler = (
            optim.lr_scheduler
            .ExponentialLR(
                optimizer,
                gamma=self.lr_multiplier
            )
        )

        loss_arr = []
        lr_arr = []

        cur_iter = 0
        best_loss = 1e9

        with tqdm(
            total=self.no_iter
        ) as pbar:

            while cur_iter < self.no_iter:

                lr_arr.append(
                    scheduler.get_lr()
                )

                cur_loss = self.train_batch(
                    model,
                    criterion,
                    optimizer,
                    scheduler,
                    dataloader
                )

                loss_arr.append(
                    cur_loss
                )

                if cur_loss < best_loss:
                    best_loss = cur_loss

                if (
                    cur_loss
                    > 2e2 * best_loss
                ):
                    print(
                        'Diverged on iteration '
                        + str(cur_iter)
                        + ' with loss '
                        + str(cur_loss)
                    )
                    break

                pbar.set_postfix(
                    loss=cur_loss
                )

                pbar.update(1)

                cur_iter += 1

        pbar.close()

        return lr_arr, loss_arr

    def grab_batch(
        self,
        dataloader
    ):
        if self.dataiter is None:
            self.dataiter = iter(
                dataloader
            )

        try:
            X, y, w = next(
                self.dataiter
            )

        except:
            self.dataiter = iter(
                dataloader
            )

            X, y, w = next(
                self.dataiter
            )

        return X, y, w

    def train_batch(
        self,
        model,
        criterion,
        optimizer,
        scheduler,
        dataloader
    ):
        optimizer.zero_grad()

        cur_iter = 0
        run_loss = 0

        while (
            cur_iter
            < self.batch_size
        ):
            X, y, w = self.grab_batch(
                dataloader
            )

            y_hat = model(X)

            loss = criterion(
                y,
                y_hat,
                w
            ) / self.batch_size

            run_loss += loss.item()

            loss.backward()

            cur_iter += 1

        optimizer.step()
        scheduler.step()

        return run_loss
```

```
    '''Early Stopping'''

    def __init__(
        self,
        patience,
        fname

        self.counter = 0

        model

                    ),
                    'loss': loss,
                },
                self.filename

            self.counter
```
